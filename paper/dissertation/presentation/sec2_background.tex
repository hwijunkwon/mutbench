%% --- Slide 1: Mutation Hotspot Detection Overview ---
\begin{frame}{Mutation Hotspot Detection in Viral Genomics}
  \begin{block}{What are mutation hotspots?}
    Genomic regions where mutations concentrate non-uniformly, reflecting sites under \highlight{structural or immunological selective pressure}
  \end{block}
  \vspace{0.5em}

  \textbf{Why hotspot detection matters:}
  \begin{enumerate}
    \item \textbf{Variant emergence prediction} --- identify positions predisposed to future mutations
    \item \textbf{Vaccine target selection} --- avoid mutable epitopes; target conserved regions
    \item \textbf{Therapeutic development} --- target structurally constrained positions
  \end{enumerate}
  \vspace{0.5em}

  RNA viruses: mutation rates of $10^{-3}$--$10^{-5}$ substitutions/nucleotide/replication cycle\\
  $\Rightarrow$ High mutation supply $+$ strong selection $=$ pronounced mutational clustering
\end{frame}

\end{frame}

%% --- Slide 3: Ground Truth Construction Challenges ---
\begin{frame}{Ground Truth Construction: Deep Mutational Scanning}
  \begin{columns}[T]
    \column{0.55\textwidth}
    \textbf{Deep Mutational Scanning (DMS):}
    \begin{itemize}
      \item Systematically measures fitness effects of \highlight{all possible} single amino acid substitutions
      \item Saturation mutagenesis $\rightarrow$ selection assay $\rightarrow$ deep sequencing
      \item Gold standard: \highlight{unbiased}, comprehensive, experimental
    \end{itemize}

    \vspace{0.5em}
    \textbf{Key DMS studies:}
    \begin{itemize}
      \item Starr et al. --- SARS-CoV-2 RBD binding
      \item Dadonaite et al. --- Full Spike entry fitness
      \item Lee et al. --- Influenza H3N2 HA
      \item Bloom \& Neher --- All SARS-CoV-2 proteins
    \end{itemize}

    \column{0.42\textwidth}
    \textbf{DMS limitations:}
    \begin{itemize}
      \item[\xmark] Expensive, labor-intensive
      \item[\xmark] Pseudovirus systems (not authentic virus)
      \item[\xmark] In vitro $\neq$ in vivo pressures
      \item[\xmark] \highlight{Sparse coverage}: EVEREST found only 45 DMS datasets across all WHO priority viruses
    \end{itemize}

    \vspace{0.5em}
    $\Rightarrow$ DMS alone cannot be universal ground truth\\
    $\Rightarrow$ Motivates \highlight{3-layer ground truth} design
  \end{columns}
\end{frame}

%% --- Slide 5: Positive Selection Detection Methods ---
\begin{frame}{Positive Selection Detection: dN/dS and Beyond}
  \small
  \begin{block}{Core concept}
    $\omega = dN/dS$: ratio of non-synonymous to synonymous substitution rates\\
    $\omega > 1$ $\Rightarrow$ \highlight{positive selection};\quad $\omega = 1$ $\Rightarrow$ neutral;\quad $\omega < 1$ $\Rightarrow$ purifying selection
  \end{block}
  \vspace{0.3em}

  \begin{columns}[T]
    \column{0.48\textwidth}
    \textbf{Codon-based methods:}
    \begin{itemize}
      \item \textbf{MEME} --- Mixed Effects Model of Evolution; detects \highlight{episodic} positive selection (subset of branches)
      \item \textbf{FUBAR} --- Bayesian per-site selection pressure estimate
      \item Both from HyPhy framework
    \end{itemize}

    \vspace{0.3em}
    \textbf{Requirements:}
    \begin{itemize}
      \item Reliable phylogenetic tree
      \item Codon-aligned sequences
      \item Not always available for all pathogens
    \end{itemize}

    \column{0.48\textwidth}
    \begin{figure}
      \centering
      \includegraphics[width=\textwidth,max height=\maxfigheight]{dnds_enrichment.png}
      \vspace{-0.3em}{\centering\scriptsize dN/dS enrichment analysis}
    \end{figure}
    \vspace{0.2em}
    In MutBench: MEME and FUBAR scores serve as information types for cross-pathogen comparison
  \end{columns}
\end{frame}

%% --- Slide 6: Density-Based Clustering and MutClust ---
\begin{frame}{Density-Based Clustering: From DBSCAN to MutClust}
  \begin{columns}[T]
    \column{0.50\textwidth}
    \textbf{DBSCAN limitations for viral data:}
    \begin{itemize}
      \item \highlight{Fixed global $\varepsilon$} cannot handle density variation
      \item SARS-CoV-2: default $\rightarrow$ 99.76\% in 1 cluster; tuned $\rightarrow$ only 4
      \item HDBSCAN: adaptive density, but no biological info integration
    \end{itemize}
    \vspace{0.3em}

    \textbf{MutClust} innovations:
    \begin{itemize}
      \item \textbf{H-score}: $H_i = \log_2(P_i \times E_i \times 100 + 1)$
      \item \textbf{Adaptive $\varepsilon$}: $\varepsilon_i = \gamma \times H_i$
      \item Detected \highlight{477 hotspots} (vs.\ DBSCAN's 4)
    \end{itemize}

    \column{0.47\textwidth}
    \textbf{MutClust limitations} $\Rightarrow$ MutBench:
    \begin{itemize}
      \item \highlight{Cumulative radius decay}: premature cluster termination
      \item \highlight{Hardcoded params}: $\gamma\!=\!10$, $d\!=\!3$ fitted to SARS-CoV-2 only
      \item \highlight{No independent GT}: self-referential bootstrap
      \item \highlight{Single info type}: only freq $+$ entropy (H-score)
    \end{itemize}
    \vspace{0.3em}
    $\Rightarrow$ Need systematic benchmark with multi-source ground truth
  \end{columns}
\end{frame}

\begin{frame}[noframenumbering]{Appendix: Key Definitions}
  \small
  \begin{description}
    \item[Mutation hotspot] Genomic region with statistically significant concentration of mutations, reflecting functional importance
    \item[H-score] $H_i = \log_2(P_i \times E_i \times 100 + 1)$; combines mutation frequency ($P$) and diversity ($E$)
    \item[Adaptive $\varepsilon$] Position-specific clustering radius: $\varepsilon_i = \gamma \times H_i$; wider for biologically significant positions
    \item[dN/dS ($\omega$)] Ratio of non-synonymous to synonymous substitution rates; $\omega > 1$ indicates positive selection
    \item[DMS] Deep Mutational Scanning --- experimental measurement of all single-residue fitness effects
    \item[PLM] Protein Language Model (e.g., ESM-2) --- learns evolutionary plausibility from sequence patterns
    \item[Hotspot-score] Composite evaluation metric balancing precision, recall, and specificity
    \item[No Free Lunch] No single algorithm outperforms all others across all problem instances $\Rightarrow$ optimal method is \highlight{pathogen-dependent}
    \item[Algorithm selection] Rice's framework: map problem features to the best-performing algorithm per instance
  \end{description}
\end{frame}
