\begin{frame}[noframenumbering]{Backup: ANOVA Robustness --- Category-Level Replication}
  \footnotesize
  \begin{block}{Concern: Collinearity Among 20 Scoring Types}
    Several scoring types share an information source (e.g., freq, freq\_with\_indel, rare\_freq). Could this inflate the interaction effect?
  \end{block}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{6-Category ANOVA (aggregated)}
        \begin{itemize}
          \item 20 scoring types grouped into 6 categories
          \item $\omega^2_{\text{cat} \times \text{path}} = \highlight{0.103}$ (medium+)
          \item Comparable to pathogen main effect ($0.117$)
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Generalization Failure}
        \begin{itemize}
          \item Friedman $\chi^2 = 7.69$, $p = 0.990$
          \item LOPO: \highlight{0/11} generalization
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \centering
  {\small \textbf{Conclusion:} Pathogen-adaptive finding is \highlight{not} an artifact of correlated scoring types.}
\end{frame}

\begin{frame}[noframenumbering]{Backup: Why Not Use Alternative Approaches?}
  \scriptsize
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Why not cancer hotspot tools?}
        \begin{itemize}
          \item Cancer tools assume somatic mutation model
          \item Viral mutations: \highlight{population-level} frequency, not clonality
        \end{itemize}
      \end{block}
      \begin{block}{Why not just use DMS?}
        \begin{itemize}
          \item DMS available for only \highlight{6/11} pathogens
          \item DMS captures function but \highlight{not} convergent evolution
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Why DBSCAN over other clustering?}
        \begin{itemize}
          \item DBSCAN: \highlight{no predefined cluster count} required
          \item Adaptive $\varepsilon$ scales with local score
        \end{itemize}
      \end{block}
      \begin{block}{Why 14 detection families?}
        \begin{itemize}
          \item \highlight{No single family dominates}: $\omega^2 = 0.013$
          \item \textit{Scoring choice} matters far more than detection
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[noframenumbering]{Backup: ESM-2 Training Data Overlap}
  \footnotesize
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{The Concern}
        \begin{itemize}
          \item ESM-2 trained on UniRef50/90 containing viral proteins
          \item All 11 proteins \highlight{likely in training corpus}
          \item LLR scores may reflect \highlight{memorized patterns}
        \end{itemize}
      \end{block}
      \begin{block}{Why This Does Not Invalidate Results}
        \begin{itemize}
          \item Concern applies \highlight{uniformly} to all 11 pathogens
          \item \highlight{Relative cross-pathogen comparisons remain valid}
          \item Main finding is interaction, not absolute performance
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Mitigating Evidence}
        \begin{itemize}
          \item ESM-2 LLR best for only \highlight{1/11} pathogens (RSV)
          \item Ranks 1st for SARS-CoV-2, last for Norovirus --- memorization would yield consistent dominance
          \item Future: evaluate PLMs trained \highlight{excluding} viral sequences
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
