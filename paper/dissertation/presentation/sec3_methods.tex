%% ---- Slide 3: 3-Layer Ground Truth ----
\begin{frame}{3-Layer Ground Truth Framework}
  \footnotesize
  \begin{columns}[T]
    \column{0.52\textwidth}
    \includegraphics[width=\textwidth,max height=0.5\textheight]{3layer_gt_concept.png}

    \column{0.46\textwidth}
    \begin{block}{Layer A -- Convergent Evolution}
      \begin{itemize}
        \item Recurrent mutations in $\geq 3$ independent lineages
        \item Role: \highlight{True Positives}; e.g., E484K
      \end{itemize}
    \end{block}
    \begin{block}{Layer B -- Purifying Selection}
      \begin{itemize}
        \item Structurally essential, dN/dS $< 0.5$
        \item Role: \highlight{True Negatives} (constrained FPR)
      \end{itemize}
    \end{block}
    \begin{block}{Layer C -- DMS Fitness}
      \begin{itemize}
        \item Top 20\% mean absolute fitness effect
        \item Role: \highlight{Independent validation} (6/11 pathogens)
      \end{itemize}
    \end{block}
  \end{columns}
\end{frame}

%% ---- Slide 4: Ground Truth Details ----
\begin{frame}{Ground Truth: Three Orthogonal Dimensions}
  \begin{block}{Design Rationale \& Key Properties}
    \begin{itemize}
      \item \textbf{Layer A}: ``Does this position mutate repeatedly and independently?'' \\
            $\rightarrow$ \textit{Evolutionary pattern} (phylogenetic convergence)
      \item \textbf{Layer B}: ``Is this position structurally constrained?'' \\
            $\rightarrow$ \textit{Structural} question (purifying selection)
      \item \textbf{Layer C}: ``What is the measurable functional impact?'' \\
            $\rightarrow$ \textit{Functional} question (DMS experiments)
      \item Each layer provides \highlight{non-redundant, independent} evidence
      \item Layer A captures signals DMS cannot; Layer C captures effects evolution has not yet revealed
    \end{itemize}
  \end{block}
\end{frame}

%% ---- Slide 8: Evaluation Metrics ----
\begin{frame}{Evaluation Metrics}
  \begin{columns}[T]
    \column{0.55\textwidth}
    \begin{block}{Primary Metric: MCC (Stage 2)}
      \vspace{-0.5em}
      \begin{equation*}
        \text{MCC} = \frac{TP \times TN - FP \times FN}{\sqrt{(TP{+}FP)(TP{+}FN)(TN{+}FP)(TN{+}FN)}}
      \end{equation*}
      \begin{itemize}
        \item Uses all 4 confusion matrix cells
        \item \highlight{Robust to class imbalance} (positive rate 0.7--15.9\%)
        \item Range: $[-1, +1]$; $0 =$ random
      \end{itemize}
    \end{block}

    \begin{block}{Hotspot-Score (Stage 1)}
      \vspace{-0.5em}
      \begin{equation*}
        \text{hotspot-score} = \frac{\text{recall} + \text{precision} + \text{stability}}{3}
      \end{equation*}
      {\small Stability = mean Jaccard similarity over 100 subsampling iterations}
    \end{block}

    \column{0.43\textwidth}
    \begin{block}{Supplementary Metrics}
      \begin{itemize}
        \item F1 (harmonic mean P/R)
        \item Precision, Recall
        \item Enrichment ratio
        \item \textbf{Constrained FPR}\\
              {\scriptsize FPR in Layer B regions}
        \item \textbf{DMS F1}\\
              {\scriptsize F1 against Layer C\\(6 pathogens only)}
      \end{itemize}
    \end{block}

    \vspace{0.3em}
    {\small Stage 1 $\rightarrow$ hotspot-score (reliability)\\
    Stage 2 $\rightarrow$ MCC (accuracy)}
  \end{columns}
\end{frame}

%% ---- Slide 9: Statistical Framework ----
\begin{frame}{Statistical Analysis Framework}
  \begin{block}{Three-Way ANOVA with $\omega^2$}
    \begin{itemize}
      \item Factors: \textbf{Scoring} (20) $\times$ \textbf{Detection} (14) $\times$ \textbf{Pathogen} (11)
      \item All main effects + all two-way interactions
      \item Partial $\omega^2$: bias-corrected variance explained ($N = 8{,}580$)
    \end{itemize}
  \end{block}

  \begin{columns}[T]
    \column{0.48\textwidth}
    \begin{block}{LOPO Cross-Validation}
      \begin{itemize}
        \item Leave-One-Pathogen-Out (11 folds)
        \item Select best scoring $\times$ detection on $n{-}1$
        \item Evaluate on held-out pathogen
        \item Random baseline: $\sim$0.36\% concordance
      \end{itemize}
    \end{block}

    \column{0.48\textwidth}
    \begin{block}{Additional Tests}
      \begin{itemize}
        \item \textbf{Friedman test}: nonparametric ranking across 11 pathogen blocks
        \item \textbf{BCa bootstrap CI}: performance variability
        \item \textbf{Subsample robustness}: 25/50/75/100\% MSA size
        \item \textbf{Kruskal--Wallis}: confirms ANOVA results
      \end{itemize}
    \end{block}
  \end{columns}
\end{frame}
