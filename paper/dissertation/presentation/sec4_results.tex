%% Slide 6: LOPO Cross-Validation
\begin{frame}{LOPO Cross-Validation: Complete Generalization Failure}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{figure}
        \includegraphics[width=\textwidth,max height=\maxfigheight]{stage3_lopo_crossval.png}
        \caption*{\footnotesize Predicted vs.\ actual best MCC per pathogen}
      \end{figure}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{block}{LOPO Results}
        \begin{itemize}
          \item Combination match: \highlight{0/11} (0\%)
          \item Mean generalization MCC: \highlight{0.032}
          \item Mean performance gap: \highlight{0.265}
        \end{itemize}
      \end{block}
      \vspace{0.3em}
      \begin{block}{Friedman Test}
        \begin{itemize}
          \item $\chi^2 = 7.69$, $p = 0.990$
          \item Kendall's $W = 0.037$
          \item \highlight{No reliably superior} combination across pathogens
        \end{itemize}
      \end{block}
      \vspace{0.3em}
      {\small $\Rightarrow$ Training-best never generalizes to a new pathogen}
    \end{column}
  \end{columns}
\end{frame}

%% Slide 12: Results Summary
\begin{frame}{Results Summary: Four Lines of Evidence}
  \small
  \begin{block}{1. Interaction Dominance}
    Scoring $\times$ pathogen: \highlight{$\omega^2 = 0.296$} --- largest variance source. \textit{Which information is used} matters most.
  \end{block}
  \begin{block}{2. No Universal Best Method}
    \highlight{11/11 unique} best combinations, 9 distinct scoring types. Friedman $p = 0.990$, $W = 0.037$.
  \end{block}
  \begin{block}{3. Generalization Failure}
    LOPO match rate: \highlight{0/11}. Mean gap = 0.265. Training-best does not transfer.
  \end{block}
  \begin{block}{4. Practical Validation}
    Vaccine escape enrichment up to \highlight{9.36-fold} ($p < 0.0001$); multi-source: 4.01-fold ($p = 0.002$).
  \end{block}
  \vspace{0.3em}
  \centering
  $\Rightarrow$ \highlight{Pathogen-adaptive method selection is necessary}, not merely beneficial.
\end{frame}
